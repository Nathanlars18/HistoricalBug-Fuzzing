Description:

torch.compile with the Inductor backend fails when compiling a model that compares
the behavior of torch.addmm and in-place Tensor.addmm_ on int64 tensors.

The same operation succeeds in eager execution but fails during fake tensor execution
inside the compiled graph.


Minimal Reproduction:

```python
import torch
import torch.nn as nn


class M(nn.Module):
    def __init__(self):
        super().__init__()

        self.mat1 = torch.tensor(
            [[1,2,3],
             [4,5,6]],
            dtype=torch.int64
        )

        self.mat2 = torch.tensor(
            [[1,0,2],
             [3,1,4],
             [5,2,6]],
            dtype=torch.int64
        )


    def forward(self, x):

        try:
            out_of_place = torch.addmm(
                x,
                self.mat1,
                self.mat2
            )

        except Exception:
            out_of_place = None


        x_inplace = x.clone()


        try:
            x_inplace.addmm_(
                self.mat1,
                self.mat2
            )

            in_place = x_inplace

        except Exception:
            in_place = None


        if out_of_place is None or in_place is None:
            return torch.tensor(False)


        return torch.equal(
            out_of_place,
            in_place
        )


model = M().eval()

x = torch.tensor(
    [[1,0,-1]],
    dtype=torch.int64
)


torch.compile(model)(x)
